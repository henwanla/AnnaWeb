<script>
(() => {
  'use strict';
  const container = document.currentScript.previousElementSibling;
  const frame = container.querySelector('iframe');
  window.addEventListener('message', event => {
    // Opaque null is expected for this sandbox; match the exact window as well.
    if (event.source !== frame.contentWindow || event.origin !== 'null') return;
    const data = event.data;
    if (!data || data.type !== 'ppi-widget-height-v1' ||
        !Number.isSafeInteger(data.height) || data.height < 200 || data.height > 2600) return;
    frame.style.height = data.height + 'px';
  });
  const url = new URL(frame.dataset.widgetSource);
  url.searchParams.set('parentOrigin', location.origin);
  frame.src = url.href;
})();
</script>
